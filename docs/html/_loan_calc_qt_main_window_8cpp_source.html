336void LoanCalcQtMainWindow::pressedButtonCalculate()
337{
338 getInputFields(); // resets the calculator and set it with input fields
339 QString result;
340
341 if(radioMonthlyPayment_->isChecked()) {
342 double payment = calculator_->calculatePayment();
343 result.append("Monthly Payment = ");
344 result.append(QString::number((double) payment, 'f', 2)); // 'f' is the float format, with 2 decimal places
345 result.append("\nTotal amt paid = ");
346 result.append(QString::number((double) (payment*calculator_->getPeriodTotal()), 'f', 2));
347
348 if(calculator_->getOpeningPercent() != 0.0 || calculator_->getOpeningFee() != 0.0)
349 {
350 result.append("\nInterest with fees = ");
351 result.append(QString::number((double) calculator_->calculateEffectiveInterestRate(), 'f', 2));
352 result.append("%");
353 }
354 }
355 else if(radioAmount_->isChecked()) {
356 result.append("Initial Loan amount = ");
357 result.append(QString::number((double) calculator_->calculateLoanAmount(), 'f', 2));
358 }
359 else if(radioInterest_->isChecked()) {
360 result.append("Yearly Interest Rate = ");
361 result.append(QString::number((double) calculator_->calculateInterestRate(), 'f', 2));
362 }
363 else if(radioNumPayments_->isChecked()) {
364 result.append("Number of payments = ");
365 result.append(QString::number((double) calculator_->calculateNumberPayments(), 'f', 2));
366 }
367 else if(radioBalance_->isChecked()) {
368 result.append("Loan Balance = ");
369 result.append(QString::number((double) calculator_->calculateLoanBalance(), 'f', 2));
370 }
371 // else this is impossible, one of them has to be selected
372
373 //textEditCalcResults_->setText(QString::fromStdString(calculator_->toString()));
374 textEditCalcResults_->setText(result);
375}
