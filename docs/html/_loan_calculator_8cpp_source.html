146/**
147 * @brief Calculate remaining loan balance after elapsed payments.
148 * @return Loan balance.
149 * @throws std::invalid_argument if amount, interest, payment, or elapsed period is not set.
150 */
151long double LoanCalculator::calculateLoanBalance() {
152 if(!amountSet_ || !interestSet_ || !periodElapsedSet_ || !paymentSet_)
153 throw invalid_argument("Must set loan amount, interest, payment, and elapsed period");
154 return (amount_*pow(1+interestPeriodic_, periodElapsed_)) - (payment_/interestPeriodic_)*(pow(1+interestPeriodic_, periodElapsed_)-1);
155}

157/**
158 * @brief Calculate monthly payment (EMI) including fees.
159 * @return Monthly payment amount.
160 * @throws std::invalid_argument if amount, interest, or total period is not set.
161 */
162long double LoanCalculator::calculatePayment() {
163 if(!amountSet_ || !interestSet_ || !periodTotalSet_)
164 throw invalid_argument("Must set loan amount, interest, and total period");
165
166 long double baseAmount = amount_ - initialPayment_;
167 long double feeAmount = openingFee_ + baseAmount * (openingPercent_/100.0);
168 long double totalAmount = baseAmount + feeAmount;
169
170 return (interestPeriodic_ * totalAmount) / (1 - pow(1+interestPeriodic_, -periodTotal_));
171}

197/**
198 * @brief Estimate yearly interest rate (approximate) based on loan parameters.
199 * @return Yearly interest rate in percent.
200 * @throws std::invalid_argument if amount, payment, or total period is not set.
201 */
202long double LoanCalculator::calculateInterestRate() {
203 if(!amountSet_ || !paymentSet_ || !periodTotalSet_)
204 throw invalid_argument("Must set amount, payment, and total period");
205
206 long double months = static_cast<long double>(periodTotal_);
207 long double q = log10(1.0 + 1.0 / months) / log10(2.0);
208 long double monthlyInterest = pow((pow(1.0 + payment_/amount_, 1.0/q) -1.0), q) -1.0;
209 return monthlyInterest*12*100;
210}

213 * @brief Calculate effective yearly interest rate including fees.

217long double LoanCalculator::calculateEffectiveInterestRate() {
218 if(!amountSet_ || !periodTotalSet_)
219 throw invalid_argument("Must set amount and total period");
220
221 long double payment = calculatePayment();
222 long double baseAmount = amount_ - initialPayment_;
223 long double months = static_cast<long double>(periodTotal_);
224 long double q = log10(1.0 + 1.0/ months) / log10(2.0);
225 long double monthlyInterest = pow((pow(1.0 + payment/baseAmount, 1.0/q) -1.0), q) -1.0;
226 return monthlyInterest*12*100;
